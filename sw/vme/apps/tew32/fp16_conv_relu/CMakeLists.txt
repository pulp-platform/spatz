find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(BUILD_TESTS AND "te16" IN_LIST VME_TILE_VARIANT)
  set(header ${CMAKE_CURRENT_BINARY_DIR}/data_cin32_h16_w16_cout64.h)
  add_custom_command(OUTPUT ${header}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/script/gen_data.py
            --output ${header}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/script/gen_data.py VERBATIM)
  add_custom_target(generate-fp16-conv-relu-latest DEPENDS ${header})
  foreach(variant IN ITEMS non-fused fused)
    set(name fp16-conv-relu-${variant}_cin32_h16_w16_cout64)
    add_snitch_test(${name} ${CMAKE_CURRENT_SOURCE_DIR}/main.c)
    set(target test-${SNITCH_TEST_PREFIX}${name})
    add_dependencies(${target} generate-fp16-conv-relu-latest)
    target_compile_definitions(${target} PRIVATE DATAHEADER="${header}")
    if(variant STREQUAL "non-fused")
      target_compile_definitions(${target} PRIVATE APP_FUSED=0)
    endif()
    target_compile_options(${target} PRIVATE -fno-vectorize -fno-slp-vectorize)
  endforeach()
endif()
